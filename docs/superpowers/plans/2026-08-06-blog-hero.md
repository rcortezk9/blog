# /blog-hero Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a manually-invoked `/blog-hero <post-slug>` skill that generates 3 Byzantine parchment/icon-style hero image candidates for a blog post via fal.ai, lets the user pick one, and wires it into the post's frontmatter and layout.

**Architecture:** A small Python package (`scripts/blog_hero/`) with pure, unit-testable modules for prompt construction, frontmatter read/write, and log-writing, plus a thin API-calling module and a CLI entrypoint (`generate.py`) that orchestrates them interactively. A Claude Code skill (`~/.claude/skills/blog-hero/SKILL.md`) invokes the CLI and drives the review/selection conversation with the user. One small Hugo template change renders the resulting `hero` frontmatter field.

**Tech Stack:** Python 3.12 (stdlib `unittest`, `argparse`, `pathlib`, `subprocess`), PyYAML, `requests`, `pass` (CLI), fal.ai REST API, Hugo/Go templates.

## Global Constraints

- Hero image size: 1200x630 (spec §Design/Generation)
- 3 candidates generated per run, from 3 distinct composition variants of the same base prompt (spec §Prompt construction)
- Default style block (used unless `--style` override is passed): "Byzantine parchment/icon aesthetic — gold, aged-paper texture, iconographic figures, muted earth-and-gold palette, consistent with an Eastern Orthodox devotional publication." (spec §Prompt construction)
- Model: `fal-ai/flux/dev` via `https://fal.run/fal-ai/flux/dev` (spec §Generation, resolved from Open Questions)
- API key stored in `pass` at `api-keys/fal-ai`, never in a plaintext file or committed to git (spec §API key setup)
- No persistent budget cap — every run prints an estimated cost and requires a yes/no confirmation before any paid API call (spec §Cost guard)
- Every completed run appends one JSON line to `static/hero/_log.jsonl` (spec §Cost guard & logging)
- Manual invocation only — this skill is never called automatically from `blog-post` or any other pipeline (spec §Non-Goals)
- No OG/Twitter variants, no multi-provider routing (spec §Non-Goals)

`scripts/blog_hero/frontmatter.py`:

```python
import re
from pathlib import Path

import yaml

FRONTMATTER_RE = re.compile(r"^---\n(.*?)\n---\n(.*)$", re.DOTALL)


class PostNotFoundError(Exception):
    pass


class FrontmatterParseError(Exception):
    pass


def _split(path):
    text = path.read_text()
    match = FRONTMATTER_RE.match(text)
    if not match:
        raise FrontmatterParseError(f"{path} has no valid YAML frontmatter block")
    frontmatter = yaml.safe_load(match.group(1)) or {}
    body = match.group(2)
    return frontmatter, body


def load_post(slug, posts_dir):
    path = Path(posts_dir) / f"{slug}.md"
    if not path.exists():
        raise PostNotFoundError(f"No post found at {path}")
    frontmatter, body = _split(path)
    return frontmatter, body, path


def set_hero_field(path, hero_relpath):
    frontmatter, body = _split(path)
    frontmatter["hero"] = hero_relpath
    new_yaml = yaml.dump(frontmatter, sort_keys=False, allow_unicode=True).strip()
    path.write_text(f"---\n{new_yaml}\n---\n{body}")
```

```python
import json
from datetime import datetime, timezone
from pathlib import Path


def append_log(log_path, slug, prompts, model, cost):
    log_path = Path(log_path)
    entry = {
        "slug": slug,
        "prompts": prompts,
        "model": model,
        "cost": cost,
        "timestamp": datetime.now(timezone.utc).isoformat(),
    }
    log_path.parent.mkdir(parents=True, exist_ok=True)
    with log_path.open("a") as f:
        f.write(json.dumps(entry) + "\n")
```

`scripts/blog_hero/fal_client.py`:

```python
import subprocess

import requests

MODEL_ID = "fal-ai/flux/dev"
API_URL = f"https://fal.run/{MODEL_ID}"
COST_PER_IMAGE = 0.025  # USD, approximate — verify against fal.ai's current pricing page


class FalApiError(Exception):
    pass


def get_api_key():
    result = subprocess.run(
        ["pass", "show", "api-keys/fal-ai"],
        capture_output=True, text=True, check=False,
    )
    if result.returncode != 0 or not result.stdout.strip():
        raise FalApiError("Could not read fal.ai API key from pass at api-keys/fal-ai")
    return result.stdout.splitlines()[0].strip()


def generate_image(prompt, api_key, width=1200, height=630):
    response = requests.post(
        API_URL,
        headers={"Authorization": f"Key {api_key}"},
        json={
            "prompt": prompt,
            "image_size": {"width": width, "height": height},
            "num_images": 1,
        },
        timeout=120,
    )
    if response.status_code != 200:
        raise FalApiError(f"fal.ai request failed: {response.status_code} {response.text}")
    data = response.json()
    images = data.get("images") or []
    if not images:
        raise FalApiError(f"fal.ai response had no images: {data}")
    image_url = images[0]["url"]
    image_response = requests.get(image_url, timeout=60)
    if image_response.status_code != 200:
        raise FalApiError(f"Could not download generated image from {image_url}")
    return image_response.content
```

```python
import argparse
import shutil
import sys
from pathlib import Path

BLOG_ROOT = Path(__file__).resolve().parents[2]
POSTS_DIR = BLOG_ROOT / "content" / "posts"
REVIEW_DIR = BLOG_ROOT / "static" / "hero" / "_review"
HERO_DIR = BLOG_ROOT / "static" / "hero"
LOG_PATH = HERO_DIR / "_log.jsonl"

sys.path.insert(0, str(BLOG_ROOT))

from scripts.blog_hero.prompt import build_prompts
from scripts.blog_hero.frontmatter import load_post, set_hero_field, PostNotFoundError, FrontmatterParseError
from scripts.blog_hero.fal_client import MODEL_ID, COST_PER_IMAGE, FalApiError, get_api_key, generate_image
from scripts.blog_hero.hero_log import append_log


def main():
    parser = argparse.ArgumentParser(description="Generate hero image candidates for a blog post.")
    parser.add_argument("slug", help="Post filename without .md, e.g. the-frame-he-remembers")
    parser.add_argument("--style", default=None, help="Override the default Byzantine style block for this run")
    args = parser.parse_args()

    try:
        frontmatter, _, post_path = load_post(args.slug, POSTS_DIR)
    except (PostNotFoundError, FrontmatterParseError) as e:
        print(f"Error: {e}")
        sys.exit(1)

    title = frontmatter.get("title", args.slug)
    description = frontmatter.get("description", "")
    tags = frontmatter.get("tags", [])

    prompts = build_prompts(title, description, tags, style_override=args.style)

    total_cost = COST_PER_IMAGE * len(prompts)
    print(f"About to generate {len(prompts)} candidates with {MODEL_ID}.")
    print(f"Estimated cost: ${total_cost:.3f}")
    confirm = input("Proceed? [y/N] ").strip().lower()
    if confirm != "y":
        print("Aborted, no API calls made.")
        sys.exit(0)

    try:
        api_key = get_api_key()
    except FalApiError as e:
        print(f"Error: {e}")
        sys.exit(1)

    REVIEW_DIR.mkdir(parents=True, exist_ok=True)
    candidate_paths = []
    for i, prompt in enumerate(prompts, start=1):
        print(f"Generating candidate {i}/{len(prompts)}...")
        try:
            image_bytes = generate_image(prompt, api_key)
        except FalApiError as e:
            print(f"Error generating candidate {i}: {e}")
            sys.exit(1)
        candidate_path = REVIEW_DIR / f"{args.slug}-{i}.jpg"
        candidate_path.write_bytes(image_bytes)
        candidate_paths.append(candidate_path)
        print(f"Saved {candidate_path}")

    append_log(LOG_PATH, args.slug, prompts, MODEL_ID, total_cost)

    print("\nCandidates ready for review:")
    for p in candidate_paths:
        print(f"  {p}")
    print("\nOpen these and tell Claude which number to keep (or discard all).")


if __name__ == "__main__":
    main()
```
